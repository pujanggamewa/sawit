    function updateBlokChart() {
    const ctx = document.getElementById('chartBlok').getContext('2d');
    if (blokChartObj) blokChartObj.destroy();
    if (!lahanData.length) return;
    const labels = lahanData.map(b => b.nama);
    const vals = lahanData.map(b => b.status === 'baru' ? 0 : parseFloat((b.pohon * 67.8 / 1000).toFixed(1)));
    const colors = lahanData.map(b => ({sehat:'#1e8a5a',perhatian:'#e8a835',kritis:'#c0392b',baru:'#1a6fa8'})[b.status] || '#1e8a5a');
    blokChartObj = new Chart(ctx, {
    type: 'bar',
    data: { labels, datasets: [{ label: 'Estimasi TBS (ton)', data: vals, backgroundColor: colors, borderRadius: 5, borderSkipped: false }] },
    options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false }, tooltip: { callbacks: { label: ctx => ` ${ctx.parsed.y} ton TBS` } } }, scales: { x: { grid: { display: false }, ticks: { font: { size: 11 } } }, y: { grid: { color: 'rgba(0,0,0,0.04)' }, ticks: { font: { size: 11 }, callback: v => v + ' t' } } } }
